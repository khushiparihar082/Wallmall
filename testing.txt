CREATE FUNCTION getThumbnailImagePath(originalImagePath VARCHAR(255))
RETURNS VARCHAR(255)
DETERMINISTIC
BEGIN
    DECLARE pathWithoutFileName VARCHAR(255);
    DECLARE fileName VARCHAR(255);
    DECLARE directoryPath VARCHAR(255);
    DECLARE thumbnailPath VARCHAR(255);

    -- Extract the directory path without the file name
    SET directoryPath = SUBSTRING_INDEX(originalImagePath, '/', -1);

    -- Extract the file name
    SET fileName = SUBSTRING_INDEX(originalImagePath, '/', -1);

    -- Extract the path before the file name
    SET pathWithoutFileName = LEFT(originalImagePath, CHAR_LENGTH(originalImagePath) - CHAR_LENGTH(fileName));

    -- Combine the path with the 'thumbnail' directory and the file name
    SET thumbnailPath = CONCAT(pathWithoutFileName, 'thumbnail/', fileName);

    RETURN thumbnailPath;
END //

DELIMITER ;

DELIMITER $$

CREATE PROCEDURE create_index_if_not_exists(
    IN p_table_name VARCHAR(255),
    IN p_index_name VARCHAR(255),
    IN p_column_list VARCHAR(255)
)
BEGIN
    DECLARE idx_count INT;

    -- Check if the index already exists
    SELECT COUNT(1) INTO idx_count
    FROM INFORMATION_SCHEMA.STATISTICS
    WHERE TABLE_SCHEMA = DATABASE()
      AND TABLE_NAME = p_table_name
      AND INDEX_NAME = p_index_name;

    -- If the index does not exist, create it
    IF idx_count = 0 THEN
        SET @create_index_stmt = CONCAT('CREATE INDEX ', p_index_name, ' ON ', p_table_name, ' (', p_column_list, ')');
        PREPARE stmt FROM @create_index_stmt;
        EXECUTE stmt;
        DEALLOCATE PREPARE stmt;
    END IF;
END$$
